def require(ok, message):
    if not ok:
        raise RuntimeError(message)


def digest(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as stream:
        for block in iter(lambda: stream.read(65536), b''):
            h.update(block)
    return h.hexdigest()


def now():
    return dt.datetime.now(dt.timezone.utc)


def timestamp(value):
    parsed = dt.datetime.fromisoformat(value)
    require(parsed.utcoffset() == dt.timedelta(0), 'explicit UTC required')
    return parsed


def load_module(path, name):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def checked_package(share):
    """Read/compile only; does not import runtime modules or touch business inputs."""
    ctx = json.loads((share/'proposal.json').read_bytes())
    require(ctx['schema'] == 'swcli.markingjig-probe-preparation.v1'
            and ctx['proposal_id'] == PROPOSAL_ID and ctx['authorized_for_CAD'] is False,
            'preparation proposal identity differs')
    require(ctx['source_head'] == SOURCE_HEAD and ctx['product_head'] == PRODUCT_HEAD,
            'reviewed source identity differs')
    require(ctx['mode'] == 'probe' and ctx['phase_policy'] == 'opening-transition-probe'
            and ctx['export_permitted'] is False, 'zero-export phase differs')
    require(ctx['guest_base'] == GUEST_BASE and ctx['guest_share'] == GUEST_SHARE,
            'fresh layout differs')
    require(ctx['approved_inventory_sha256'] == INVENTORY_SHA
            and ctx['launcher_receipts_sha256'] == HELPER_SHA, 'inventory/helper pin differs')
    require(set(ctx['code_files_sha256']) == set(MEMBERS), 'runtime member list differs')
    for name, expected in ctx['preparation_files_sha256'].items():
        require(name in ('launch-probe.py', 'launcher_receipts.py', 'query-context.py', 'host-original-guard.py'),
                'unknown preparation file')
        require(digest(share/name) == expected, 'preparation file differs: '+name)
        compile((share/name).read_bytes(), name, 'exec')
    require(set(ctx['preparation_files_sha256']) ==
            {'launch-probe.py', 'launcher_receipts.py', 'query-context.py', 'host-original-guard.py'},
            'preparation source missing')
    require(digest(share/'launcher_receipts.py') == HELPER_SHA, 'standalone helper differs')
    packed_path = share/'markingjig-code-c9846f6.zip'
    require(digest(packed_path) == ctx['package_sha256'], 'code ZIP differs')
    with zipfile.ZipFile(packed_path) as packed:
        rows = packed.infolist()
        require(len(rows) == len(MEMBERS) and set(packed.namelist()) == set(MEMBERS),
                'duplicate or unexpected ZIP members')
        for row in rows:
            p = PurePosixPath(row.filename)
            require(not p.is_absolute() and '..' not in p.parts and '\\' not in row.filename
                    and not row.is_dir() and row.file_size <= 2*1024*1024,
                    'unsafe code member')
            require(not stat.S_ISLNK(row.external_attr >> 16), 'symlink code member')
            data = packed.read(row)
            require(hashlib.sha256(data).hexdigest() == ctx['code_files_sha256'][row.filename],
                    'code file differs: '+row.filename)
            if row.filename.endswith('.py'):
                compile(data, row.filename, 'exec')
        require(ctx['code_files_sha256'][CONTROL+'launcher_receipts.py'] == HELPER_SHA,
                'supervisor receipt dependency differs')
        inv = json.loads(packed.read(INVENTORY))

    require(type(ctx['selected']) is list and len(ctx['selected']) == 10
            and len(set(ctx['selected'])) == 10 and set(ctx['selected']) == set(inv['files']),
            'exact ten approved inputs required')

def checked_grant(share, ctx, grant, current=None):
    """No execution or file writes. A proposal alone never authorizes CAD."""
    current = current or now()

            and grant.get('authorized_for_CAD') is True, 'new coordinator CAD grant required')
    ticket = grant.get('allocation_id', '')
    require(type(ticket) is str and ticket.startswith('VM-') and 'MARKINGJIG' in ticket
            and ticket.endswith('-03'), 'new MarkingJig CAD allocation required')
    require(grant.get('reviewer_thread_id') == '01a0bfca-5e87-7e61-ab5f-0ef249443bab',
            'coordinator review required')
    require(grant.get('proposal_sha256') == digest(share/'proposal.json')
            and grant.get('launcher_sha256') == digest(share/'launch-probe.py'),
            'grant does not bind this complete proposal/launcher')
    require(grant.get('package_sha256') == ctx['package_sha256']
            and grant.get('launcher_receipts_sha256') == HELPER_SHA,
            'grant package/helper differs')
    age = (current-timestamp(grant['issued_utc'])).total_seconds()
    require(0 <= age <= 300 and current < timestamp(grant['expires_utc'])
            <= timestamp(grant['issued_utc'])+dt.timedelta(minutes=10), 'grant is not fresh')

            and type(grant.get('expected_user')) is str and grant['expected_user']

            'reviewed interactive SID/user/session required')

    proof_path = share/'host-originals-before.json'
    require(digest(proof_path) == grant['host_originals_before_sha256'], 'host original proof differs')
    proof = json.loads(proof_path.read_bytes())

    return proof


def safe_local_directory(path):
    for parent in (path, *path.parents):
        if os.path.lexists(parent):
            info = parent.lstat()
            require(stat.S_ISDIR(info.st_mode) and not stat.S_ISLNK(info.st_mode)
                    and not getattr(info, 'st_file_attributes', 0) & 0x400,
                    'private directory symlink/reparse refused')

def write_entry_arrival(share):
    """One OS-only arrival record before reading proposal/grant; never authorizes CAD."""
    require(sys.platform == 'win32', 'Windows entry observation required')
    row = dict(pid=os.getpid(), argv=list(sys.argv), entry_source_sha256=digest(Path(__file__)),
               utc=now().isoformat(), session_query={})
    try:
        kernel = ctypes.WinDLL('kernel32', use_last_error=True)
        kernel.ProcessIdToSessionId.argtypes = [ctypes.c_ulong, ctypes.POINTER(ctypes.c_ulong)]
        kernel.ProcessIdToSessionId.restype = ctypes.c_int
        session = ctypes.c_ulong()
        ctypes.set_last_error(0)
        ok = kernel.ProcessIdToSessionId(os.getpid(), ctypes.byref(session))
        row['session_query'] = dict(ok=bool(ok), last_error=ctypes.get_last_error(),
                                    session=session.value if ok else None)
    except Exception as error:
        row['session_query'] = dict(ok=False, type=type(error).__name__, error=str(error))
    with (share/'entry-arrived.json').open('x', encoding='utf-8') as stream:
        json.dump(row, stream, indent=2); stream.write('\n')


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--check-package', action='store_true')
    parser.add_argument('--execute', action='store_true')
    args = parser.parse_args(argv)
    require(not (args.check_package and args.execute), 'choose check or explicit execution')
    share = Path(__file__).resolve().parent
    if args.execute:
        write_entry_arrival(share)
    ctx, inv = checked_package(share)
    if not args.execute:
        print(json.dumps(dict(package_valid=True, native_operations=0, source_head=SOURCE_HEAD,
                              proposal_id=PROPOSAL_ID, runtime_members=len(MEMBERS))))
        return 0
    grant = json.loads((share/'coordinator-cad-grant.json').read_bytes())
    checked_grant(share, ctx, grant)
    return execute(share, ctx, inv, grant)
